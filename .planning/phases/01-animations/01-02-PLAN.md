
    Use GSAP ScrollTrigger.batch for the grid card reveals:
    ```tsx
    'use client'
    import { useEffect, useRef } from 'react'
    import { gsap, ScrollTrigger } from '@/lib/gsap'
    import { FadeIn } from '@/components/animations'

    export function FeaturesSection({ features }: FeaturesSectionProps) {
      const gridRef = useRef<HTMLDivElement>(null)

      useEffect(() => {
        if (!gridRef.current || typeof window === 'undefined') return
        const cards = gridRef.current.querySelectorAll('.feature-card')
        // Set initial state
        gsap.set(cards, { opacity: 0, y: 40 })
        const ctx = gsap.context(() => {
          ScrollTrigger.batch(cards, {
            onEnter: (batch) => {
              gsap.to(batch, {
                opacity: 1,
                y: 0,
                duration: 0.6,
                ease: 'power2.out',
                stagger: 0.1,
              })
            },
            start: 'top 88%',
            once: true,
          })
        })
        return () => ctx.revert()
      }, [])

      // Add className="feature-card" to each card div
      return (
        <section className="py-24 px-6 bg-white">
          {/* section heading wrapped in FadeIn */}
          <FadeIn direction="up" className="text-center mb-16">
            {/* exact heading JSX from page.tsx */}
          </FadeIn>
          <div ref={gridRef} className="grid sm:grid-cols-2 lg:grid-cols-3 gap-6">
            {features.map((f) => (
              <div key={f.title} className="feature-card bg-gray-50 rounded-2xl p-8 ...">
                {/* exact feature card JSX */}
              </div>
            ))}
          </div>
        </section>
      )
    }
    ```
