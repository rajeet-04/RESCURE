    ```ts
    'use client'
    // Single registration point for GSAP plugins.
    // Import from here instead of 'gsap' directly to ensure plugins are registered.
    import { gsap } from 'gsap'
    import { ScrollTrigger } from 'gsap/ScrollTrigger'

    if (typeof window !== 'undefined') {
      gsap.registerPlugin(ScrollTrigger)
    }

    export { gsap, ScrollTrigger }
    ```

    Step C — Append to `src/app/globals.css` (inside @layer utilities or after existing rules):
    ```css
    /* Animation performance — applied to elements that will transform */
    .will-animate {
      will-change: transform, opacity;
    }
    .will-animate-transform {
      will-change: transform;
    }
    /* Reduce motion: honour OS-level preference */
    @media (prefers-reduced-motion: reduce) {
      *, *::before, *::after {
        animation-duration: 0.01ms !important;
        animation-iteration-count: 1 !important;
        transition-duration: 0.01ms !important;
      }
    }
    /* Clip container for text-reveal overflow */
    .clip-text {
      overflow: hidden;
      display: block;
    }
    ```

    Do NOT use `tailwindcss-animate` keyframes for these — the CSS above is for GSAP/Framer targets.

    Create each file as follows. All files start with `'use client'`.

    ─── fade-in.tsx ────────────────────────────────────────────────────────
    Framer Motion component. Props: children, delay (default 0), duration (default 0.6),
    direction ('up'|'down'|'left'|'right', default 'up'), amount (viewport threshold 0-1, default 0.15),
    once (boolean, default true), className.

    Use `motion.div` with:
    - `initial`: opacity 0 + translate offset 40px in direction  
    - `whileInView`: opacity 1 + translate 0  
    - `viewport`: `{ once, amount }`  
    - `transition`: `{ duration, delay, ease: 'easeOut' }`

    ─── stagger-children.tsx ───────────────────────────────────────────────
    Two exports: `StaggerChildren` (container) + `StaggerItem` (child).

    Container variant:
    - `initial="hidden"`, `whileInView="visible"`, `viewport={{ once: true, amount: 0.1 }}`
    - variant `hidden`: `{}`
    - variant `visible`: `{ transition: { staggerChildren: staggerDelay (default 0.12), delayChildren: delay (default 0) } }`

    Child variant:
    - variant `hidden`: `{ opacity: 0, y: 30 }`  
    - variant `visible`: `{ opacity: 1, y: 0, transition: { duration: 0.5, ease: 'easeOut' } }`

    Props for StaggerChildren: children, staggerDelay (number, default 0.12), delay (number, default 0), className.

    ─── parallax-layer.tsx ─────────────────────────────────────────────────
    GSAP ScrollTrigger scrub parallax. Props: children, speed (number, default 0.3 — fraction of
    scroll distance to move; negative = moves opposite), className.

    Implementation:
    ```tsx
    const ref = useRef<HTMLDivElement>(null)
    useEffect(() => {
      const el = ref.current
      if (!el || typeof window === 'undefined') return
      const ctx = gsap.context(() => {
        gsap.to(el, {
          yPercent: speed * 100,   // e.g. speed=0.3 → moves 30% of its height over scrub range
          ease: 'none',
          scrollTrigger: {
            trigger: el,
            start: 'top bottom',
            end: 'bottom top',
            scrub: true,
          },
        })
      })
      return () => ctx.revert()
    }, [speed])
    ```
    Wrap children in `<div ref={ref} className={className}>`.
    Import `{ gsap, ScrollTrigger }` from `@/lib/gsap`.

    ─── count-up.tsx ───────────────────────────────────────────────────────
    GSAP counter. Props: to (number), duration (default 1.5), className, prefix (string, default ''),
    suffix (string, default ''), separator (boolean — add comma/locale formatting, default true).

    Implementation:
    ```tsx
    const ref = useRef<HTMLSpanElement>(null)
    useEffect(() => {
      const el = ref.current
      if (!el || typeof window === 'undefined') return
      const ctx = gsap.context(() => {
        const obj = { val: 0 }
        gsap.to(obj, {
          val: to,
          duration,
          ease: 'power2.out',
          scrollTrigger: {
            trigger: el,
            start: 'top 85%',
            once: true,
          },
          onUpdate() {
            el.textContent = prefix +
              (separator ? Math.round(obj.val).toLocaleString('en-IN') : Math.round(obj.val).toString()) +
              suffix
          },
        })
      })
      return () => ctx.revert()
    }, [to, duration, prefix, suffix, separator])
    ```
    Render `<span ref={ref} className={className}>0</span>`.

    ─── text-reveal.tsx ────────────────────────────────────────────────────
    Word-by-word Framer Motion cascade. Props: text (string), as ('h1'|'h2'|'h3'|'p', default 'p'),
    delay (default 0), staggerDelay (default 0.05), className.

    Split `text` by spaces into words. Render each word in a `<span className="clip-text inline-block">`.
    Inside each span, a `motion.span` with:
    - `initial={{ y: '110%', opacity: 0 }}`  
    - `whileInView={{ y: 0, opacity: 1 }}`  
    - `viewport={{ once: true, amount: 0.5 }}`  
    - `transition={{ duration: 0.5, delay: delay + index * staggerDelay, ease: [0.33, 1, 0.68, 1] }}`
    
    Join words with space (use `&nbsp;` between spans).
    Render using the `as` prop as the outer tag.

    ─── index.ts ────────────────────────────────────────────────────────────
    Barrel export:
    ```ts
    export { FadeIn } from './fade-in'

    export { ParallaxLayer } from './parallax-layer'
    export { CountUp } from './count-up'
    export { TextReveal } from './text-reveal'
    ```
