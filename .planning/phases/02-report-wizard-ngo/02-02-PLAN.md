    ```typescript
    import { GoogleGenerativeAI, HarmBlockThreshold, HarmCategory } from '@google/generative-ai'
    import type { UrgencyLevel } from './urgency-scorer'

    export interface AIReportAnalysis {
      urgency: UrgencyLevel
      confidence: number
      reasoning: string
      injuryDescription: string
      recommendedAction: string
      // New fields for wizard prefill:
      animalType: 'Dog' | 'Cat' | 'Cow' | 'Other'
      suggestedTitle: string       // e.g. "Injured dog with broken leg"
      suggestedDescription: string // 2-3 sentence description for the textarea
      estimatedAge?: string        // e.g. "Adult", "Puppy/Kitten", "Senior"
      raw?: unknown
    }

    const ANALYSIS_PROMPT = `You are a veterinary triage AI for RESCURE, a stray animal rescue platform in India.

    Analyze the provided image of a stray or injured animal. Respond ONLY with valid JSON matching this exact schema — no markdown, no explanation outside JSON:

    {
      "urgency": "CRITICAL" | "HIGH" | "MEDIUM" | "LOW",
      "confidence": <0-1 float>,
      "reasoning": "<1-2 sentence triage justification>",
      "injuryDescription": "<brief description of visible condition>",
      "recommendedAction": "<immediate action for rescue team>",
      "animalType": "Dog" | "Cat" | "Cow" | "Other",
      "suggestedTitle": "<10-60 char title describing the emergency, e.g. 'Injured dog with bleeding wound'>",
      "suggestedDescription": "<2-3 sentences describing situation, condition, and why rescue is needed>",
      "estimatedAge": "Puppy/Kitten" | "Young" | "Adult" | "Senior" | null
    }

    Urgency:
    - CRITICAL: life-threatening (severe bleeding, unconscious, broken limb, cannot breathe, burns, trauma)
    - HIGH: significant injury needing same-day rescue (deep wounds, severe limping, infection, newborn, trapped)
    - MEDIUM: injured but stable (minor wounds, mange, malnourished, sick but conscious)
    - LOW: vulnerable but not injured (thin, needs food/shelter, no visible injury)

    For Indian context: stray dogs and cats are most common. Cows are protected animals — always HIGH+.`

    export async function analyzeReport(
      imageBase64: string,
      mimeType: 'image/jpeg' | 'image/png' | 'image/webp' = 'image/jpeg'
    ): Promise<AIReportAnalysis> {
      const genAI = new GoogleGenerativeAI(process.env.GOOGLE_AI_API_KEY!)

      const model = genAI.getGenerativeModel({
        model: 'gemini-2.0-flash',
        safetySettings: [
          { category: HarmCategory.HARM_CATEGORY_HARASSMENT, threshold: HarmBlockThreshold.BLOCK_NONE },
          { category: HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT, threshold: HarmBlockThreshold.BLOCK_NONE },
        ],
      })

      const result = await model.generateContent([
        ANALYSIS_PROMPT,
        { inlineData: { mimeType, data: imageBase64 } },
      ])

      const text = result.response.text().trim()
      const jsonStr = text.replace(/^```json?\n?/, '').replace(/\n?```$/, '').trim()

      try {
        const parsed = JSON.parse(jsonStr) as AIReportAnalysis
        return { ...parsed, raw: result.response }
      } catch {
        return {
          urgency: 'MEDIUM',
          confidence: 0.3,
          reasoning: 'AI parsing failed — defaulted to MEDIUM.',
          injuryDescription: 'Unable to parse AI response',
          recommendedAction: 'Manual triage required',
          animalType: 'Other',
          suggestedTitle: 'Animal needs help',
          suggestedDescription: 'An animal was spotted and appears to need rescue assistance.',
          estimatedAge: null,
          raw: text,
        }
      }
    }
    ```

    Create `src/app/api/incidents/[id]/analyze/route.ts`.

    This endpoint is called by the client after Step 1 photo submission while showing the "Analyzing..." spinner.
    It fetches the Cloudinary image, runs analyzeReport, updates the incident, and returns prefill JSON.

    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { prisma } from '@/lib/prisma'
    import { analyzeReport } from '@/lib/ai/report-analyzer'

    export async function POST(
      req: NextRequest,
      { params }: { params: Promise<{ id: string }> }
    ) {
      const { id } = await params

      try {
        const incident = await prisma.incidentReport.findUnique({
          where: { id },
          select: { id: true, photos: true, urgencyScore: true },
        })

        if (!incident) {
          return NextResponse.json({ error: 'Incident not found' }, { status: 404 })
        }

        const imageUrl = incident.photos[0] ?? null

        if (!imageUrl) {
          return NextResponse.json({
            urgency: 'MEDIUM',
            confidence: 0,
            animalType: 'Other',
            suggestedTitle: '',
            suggestedDescription: '',
            injuryDescription: '',
            recommendedAction: 'Please describe the situation manually.',
          })
        }

        // Fetch image from Cloudinary and convert to base64
        const imgRes = await fetch(imageUrl, { signal: AbortSignal.timeout(10000) })
        if (!imgRes.ok) throw new Error('Could not fetch image')

        const arrayBuffer = await imgRes.arrayBuffer()
        const base64 = Buffer.from(arrayBuffer).toString('base64')
        const mimeType = (imgRes.headers.get('content-type') ?? 'image/jpeg') as 'image/jpeg' | 'image/png' | 'image/webp'

        const analysis = await analyzeReport(base64, mimeType)

        // Update incident with all AI fields in one write
        await prisma.incidentReport.update({
          where: { id },
          data: {
            urgencyScore: analysis.urgency,
            aiRawResponse: { reasoning: analysis.reasoning, confidence: analysis.confidence },
            // Use suggestedTitle/Description as defaults if user hasn't filled them yet
            title: analysis.suggestedTitle || undefined,
            animalType: analysis.animalType || undefined,
          },
        })

        return NextResponse.json({
          urgency: analysis.urgency,
          confidence: analysis.confidence,
          animalType: analysis.animalType,
          suggestedTitle: analysis.suggestedTitle,
          suggestedDescription: analysis.suggestedDescription,
          injuryDescription: analysis.injuryDescription,
          recommendedAction: analysis.recommendedAction,
          estimatedAge: analysis.estimatedAge ?? null,
        })
      } catch (err) {
        console.error('[POST /api/incidents/[id]/analyze]', err)
        // Return graceful fallback so wizard can still proceed
        return NextResponse.json({
          urgency: 'MEDIUM',
          confidence: 0,
          animalType: 'Other',
          suggestedTitle: '',
          suggestedDescription: '',
          injuryDescription: 'Analysis unavailable.',
          recommendedAction: 'Team will assess on arrival.',
        })
      }
    }
    ```
