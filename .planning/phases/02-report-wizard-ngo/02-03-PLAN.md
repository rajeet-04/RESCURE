<objective>
Build the automatic NGO notification pipeline that runs fire-and-forget after every incident report:
1. Scrape Google Maps for nearby animal shelters
2. Query DB for registered NGOs with matching coverage zone
3. For scraped shelters: create Bitrix24 CRM lead + log activity
4. For DB NGOs: push notification + Resend email
5. Log all attempts to NGOOutreach table

Purpose: Zero manual NGO alerting — every report automatically contacts the nearest help.
Output: POST /api/incidents/[id]/notify-external
</objective>

    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { prisma } from '@/lib/prisma'
    import { findNearbyAnimalShelters } from '@/lib/geo/maps-scraper'
    import { createNGOLead, logCallActivity } from '@/lib/bitrix24/client'
    import { getNeighborHashes, DEDUP_PRECISION } from '@/lib/geo/geohash'
    import { resend } from '@/lib/resend'
    import webpush from 'web-push'
    import ngeohash from 'ngeohash'

    webpush.setVapidDetails(
      process.env.VAPID_SUBJECT!,
      process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY!,
      process.env.VAPID_PRIVATE_KEY!
    )

    export async function POST(
      req: NextRequest,
      { params }: { params: Promise<{ id: string }> }
    ) {
      // Internal key guard
      if (req.headers.get('x-internal-key') !== process.env.INTERNAL_API_KEY) {
        return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
      }

      const { id } = await params

      try {
        const incident = await prisma.incidentReport.findUnique({
          where: { id },
          select: { id: true, lat: true, lng: true, geohash: true, description: true, urgencyScore: true, title: true },
        })
        if (!incident) return NextResponse.json({ error: 'Not found' }, { status: 404 })

        const incidentDesc = incident.description ?? incident.title ?? 'Animal emergency reported on RESCURE'

        // ── 1. Google Maps scraper ─────────────────────────────────────────────
        const externalShelters = await findNearbyAnimalShelters(incident.lat, incident.lng, 15)

        // ── 2. DB-registered NGOs with matching coverage zone ─────────────────
        const geohash5 = incident.geohash.slice(0, DEDUP_PRECISION)
        const neighborHashes = getNeighborHashes(geohash5)
        const nearbyNGOs = await prisma.nGO.findMany({
          where: {
            verified: true,
            coverageZones: { some: { geohash: { in: neighborHashes } } },
          },
          include: {
            coverageZones: false,
            user: { select: { email: true, name: true } },
            fieldWorkers: { where: { available: true }, select: { userId: true } },
          },
        })

        const outreachRecords: Array<Parameters<typeof prisma.nGOOutreach.create>[0]['data']> = []

        // ── 3. Contact external shelters via Bitrix24 ─────────────────────────
        for (const shelter of externalShelters) {
          let bitrix24LeadId: string | null = null
          let status = 'pending'
          let errorMsg: string | null = null

          try {
            if (shelter.phone || shelter.website) {
              bitrix24LeadId = await createNGOLead({
                shelterName: shelter.name,
                phone: shelter.phone,
                website: shelter.website,
                address: shelter.address,
                incidentId: incident.id,
                incidentDescription: incidentDesc,
                urgency: incident.urgencyScore,
                lat: incident.lat,
                lng: incident.lng,
              })

              if (bitrix24LeadId && shelter.phone) {
                await logCallActivity({
                  leadId: bitrix24LeadId,
                  shelterName: shelter.name,
                  phone: shelter.phone,
                  incidentId: incident.id,
                })
              }

              status = bitrix24LeadId ? 'sent' : 'failed'
            } else {
              status = 'failed'
              errorMsg = 'No phone or website available for contact'
            }
          } catch (e) {
            status = 'failed'
            errorMsg = String(e)
          }

          outreachRecords.push({
            incidentId: incident.id,
            name: shelter.name,
            address: shelter.address || null,
            phone: shelter.phone,
            website: shelter.website,
            lat: shelter.lat,
            lng: shelter.lng,
            source: 'google_maps',
            contactType: 'bitrix24_lead',
            status,
            bitrix24LeadId,
            errorMsg,
            sentAt: status === 'sent' ? new Date() : null,
          })
        }

        // ── 4. Contact registered DB NGOs ─────────────────────────────────────
        for (const ngo of nearbyNGOs) {
          let status = 'pending'
          let errorMsg: string | null = null

          try {
            // Push notification to all available workers
            for (const worker of ngo.fieldWorkers) {
              const subs = await prisma.pushSubscription.findMany({
                where: { userId: worker.userId },
              })
              for (const sub of subs) {
                webpush.sendNotification(
                  JSON.parse(sub.endpoint) as Parameters<typeof webpush.sendNotification>[0],
                  JSON.stringify({
                    title: `🚨 New ${incident.urgencyScore} incident nearby`,
                    body: incidentDesc.slice(0, 100),
                    url: `/cases`,
                  })
                ).catch(() => {})
              }
            }

            // Email to NGO owner
            if (ngo.user.email) {
              await resend.emails.send({
                from: process.env.EMAIL_FROM!,
                to: ngo.user.email,
                subject: `[RESCURE] 🚨 ${incident.urgencyScore} Animal Emergency — Action Required`,
                html: `
                  <h2>Animal Emergency Reported Near Your Coverage Zone</h2>
                  <p><strong>Urgency:</strong> ${incident.urgencyScore}</p>
                  <p><strong>Location:</strong> ${incident.lat.toFixed(5)}, ${incident.lng.toFixed(5)}</p>
                  <p><strong>Description:</strong> ${incidentDesc}</p>
                  <p><strong>Incident ID:</strong> ${incident.id}</p>
                  <p><a href="${process.env.NEXT_PUBLIC_APP_URL}/cases">View in Dashboard →</a></p>
                  <hr/>
                  <small>Sent automatically by RESCURE. Please log in to assign a case worker.</small>
                `,
              })
              status = 'sent'
            }
          } catch (e) {
            status = 'failed'
            errorMsg = String(e)
          }

          outreachRecords.push({
            incidentId: incident.id,
            name: ngo.name,
            address: null,
            phone: null,
            email: ngo.user.email ?? null,
            website: null,
            source: 'db',
            contactType: 'email',
            status,
            sentAt: status === 'sent' ? new Date() : null,
            errorMsg,
          })
        }

        // ── 5. Bulk insert all outreach records ────────────────────────────────
        if (outreachRecords.length > 0) {
          await prisma.nGOOutreach.createMany({ data: outreachRecords as never, skipDuplicates: true })
        }

        return NextResponse.json({
          externalContacted: externalShelters.length,
          dbNGOsContacted: nearbyNGOs.length,
          totalOutreach: outreachRecords.length,
        })
      } catch (err) {
        console.error('[notify-external]', err)
        return NextResponse.json({ error: 'Pipeline failed' }, { status: 500 })
      }
    }
    ```

    Note: The `PushSubscription.endpoint` field stores the full subscription JSON stringified — parse accordingly. If it stores just the endpoint URL, adjust accordingly by reading the push/server.ts pattern.
