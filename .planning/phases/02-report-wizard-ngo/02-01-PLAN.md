    Create `src/lib/geo/maps-scraper.ts`. This parses the exact response format seen in `data/f.txt`.

    **Response format facts (verified from data/f.txt)**:
    - Response starts with `)]}'\n` XSSI prefix — strip it before JSON.parse
    - Top-level is a deeply nested array; place entries are within it
    - Each place entry contains pattern: `[null,"SESSION_TOKEN","QUERY_TOKEN",["addr_part1","addr_part2","full_addr","city, state"],null,[null,null,null,[..reviews..],...],...website_arr...,null,[null,null,LAT,LNG],"PLACE_TOKEN",...,null,PLACE_NAME_TOKEN,...,null,[...phone_arr...],...,null,[[[7,[["NAME"],...]]...]],...]`
    - Phone pattern: `[["DISPLAY_PHONE",1],["+91 DISPLAY",2]],null,"RAWDIGITS",null,["tel:RAWDIGITS"`
    - Website pattern: `["http://URL","domain.com",null,null,`
    - Lat/Lng appears near the END of each major entry as `[null,null,LAT,LNG]`
    - Name appears inside nested array at position: `[[[7,[["NAME"],["addr"]...]]` form

    **Implementation**:

    ```typescript
    'use server' // mark for server-only — never bundle to client

    export interface ExternalShelter {
      name: string
      address: string
      phone: string | null
      website: string | null
      lat: number
      lng: number
      placeId: string | null
    }

    const MAPS_TIMEOUT_MS = 8000

    export async function findNearbyAnimalShelters(
      lat: number,
      lng: number,
      radiusKm = 15
    ): Promise<ExternalShelter[]> {
      // Build the search URL — center on actual incident coordinates
      // The pb= param encodes the search viewport; use a simple version
      const q = encodeURIComponent('animal shelter near me')
      const url = `https://www.google.com/search?tbm=map&authuser=0&hl=en&gl=in&q=${q}&near=${lat},${lng}`

      const controller = new AbortController()
      const timer = setTimeout(() => controller.abort(), MAPS_TIMEOUT_MS)

      let raw: string
      try {
        const res = await fetch(url, {
          signal: controller.signal,
          headers: {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36',
            'Accept-Language': 'en-US,en;q=0.9',
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
            'Referer': 'https://www.google.com/',
          },
        })
        if (!res.ok) return []
        raw = await res.text()
      } catch {
        return []
      } finally {
        clearTimeout(timer)
      }

      return parseMapResults(raw, lat, lng, radiusKm)
    }

    function parseMapResults(raw: string, originLat: number, originLng: number, radiusKm: number): ExternalShelter[] {
      // Strip XSSI prefix
      const jsonStart = raw.indexOf('[')
      if (jsonStart === -1) return []
      const jsonStr = raw.slice(jsonStart)

      let data: unknown
      try {
        data = JSON.parse(jsonStr)
      } catch {
        // Fallback: regex extraction from raw string
        return extractByRegex(raw, originLat, originLng, radiusKm)
      }

      return extractByRegex(raw, originLat, originLng, radiusKm)
    }

    function haversineKm(lat1: number, lng1: number, lat2: number, lng2: number): number {
      const R = 6371
      const dLat = ((lat2 - lat1) * Math.PI) / 180
      const dLng = ((lng2 - lng1) * Math.PI) / 180
      const a =
        Math.sin(dLat / 2) ** 2 +
        Math.cos((lat1 * Math.PI) / 180) *
          Math.cos((lat2 * Math.PI) / 180) *
          Math.sin(dLng / 2) ** 2
      return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
    }

    function extractByRegex(raw: string, originLat: number, originLng: number, radiusKm: number): ExternalShelter[] {
      const results: ExternalShelter[] = []
      const seen = new Set<string>()

      // Extract all lat/lng pairs: [null,null,LAT,LNG]
      const coordPattern = /\[null,null,([\d.]+),([\d.]+)\]/g
      // Extract all phone numbers: "tel:DIGITS"
      const phonePattern = /"tel:(\d{7,15})"/g
      // Extract websites: ["http(s)://URL","domain"
      const websitePattern = /\["(https?:\/\/[^"]{5,200})","([^"]{3,50})"/g
      // Extract names from [[7,[["NAME"] pattern
      const namePattern = /\[\[\[7,\[\["([^"]{3,80})"/g

      // Collect all coords
      const coords: Array<{lat: number; lng: number; idx: number}> = []
      let m: RegExpExecArray | null
      while ((m = coordPattern.exec(raw)) !== null) {
        const la = parseFloat(m[1])
        const lo = parseFloat(m[2])
        if (la > 5 && la < 35 && lo > 65 && lo < 100) { // India bounding box
          coords.push({ lat: la, lng: lo, idx: m.index })
        }
      }

      // Collect all phones
      const phones: Array<{digits: string; idx: number}> = []
      while ((m = phonePattern.exec(raw)) !== null) {
        phones.push({ digits: m[1], idx: m.index })
      }

      // Collect all websites
      const websites: Array<{url: string; idx: number}> = []
      while ((m = websitePattern.exec(raw)) !== null) {
        if (!m[1].includes('google.com') && !m[1].includes('business.site') === false) {
          // keep including business.site but skip google-owned domains except business.site
        }
        if (!m[1].includes('google.com') && !m[1].includes('gstatic.com') && !m[1].includes('googleusercontent.com') && !m[1].includes('googleapis.com')) {
          websites.push({ url: m[1], idx: m.index })
        }
      }

      // Collect all names
      const names: Array<{name: string; idx: number}> = []
      while ((m = namePattern.exec(raw)) !== null) {
        if (m[1].length > 3 && !m[1].startsWith('http')) {
          names.push({ name: m[1], idx: m.index })
        }
      }

      // Match coords to closest name + phone + website by index proximity
      for (const coord of coords) {
        if (haversineKm(originLat, originLng, coord.lat, coord.lng) > radiusKm) continue
        const key = `${coord.lat.toFixed(4)},${coord.lng.toFixed(4)}`
        if (seen.has(key)) continue
        seen.add(key)

        // Find closest name appearing BEFORE this coord
        const closestName = names
          .filter(n => n.idx < coord.idx)
          .sort((a, b) => b.idx - a.idx)[0]

        // Find closest phone within ±5000 chars of coord
        const closestPhone = phones
          .filter(p => Math.abs(p.idx - coord.idx) < 5000)
          .sort((a, b) => Math.abs(a.idx - coord.idx) - Math.abs(b.idx - coord.idx))[0]

        // Find closest website within ±8000 chars
        const closestWebsite = websites
          .filter(w => Math.abs(w.idx - coord.idx) < 8000)
          .sort((a, b) => Math.abs(a.idx - coord.idx) - Math.abs(b.idx - coord.idx))[0]

        if (!closestName) continue

        results.push({
          name: closestName.name,
          address: '',    // address extraction done separately below
          phone: closestPhone?.digits ?? null,
          website: closestWebsite?.url ?? null,
          lat: coord.lat,
          lng: coord.lng,
          placeId: null,
        })
      }

      // Extract addresses: pattern comes from the address_parts array near each entry
      // Look for pattern: "Place Name, Street, Area, City, State PINCODE"
      const addrPattern = /"([^"]{10,200},[^"]{5,100},[^"]{5,100})"/g
      const addrs: Array<{addr: string; idx: number}> = []
      while ((m = addrPattern.exec(raw)) !== null) {
        if (m[1].includes('West Bengal') || m[1].includes('Maharashtra') || m[1].includes('Karnataka') ||
            m[1].includes('Tamil Nadu') || m[1].includes('Delhi') || m[1].includes('Gujarat') ||
            m[1].match(/\d{6}/)) {
          addrs.push({ addr: m[1], idx: m.index })
        }
      }

      // Back-fill address for each result
      for (const r of results) {
        const coordIdx = coords.find(c => c.lat === r.lat && c.lng === r.lng)?.idx ?? 0
        const closest = addrs
          .filter(a => Math.abs(a.idx - coordIdx) < 10000)
          .sort((a, b) => Math.abs(a.idx - coordIdx) - Math.abs(b.idx - coordIdx))[0]
        if (closest) r.address = closest.addr
      }

      return results.slice(0, 10) // cap at 10 results
    }
    ```

    ENV vars needed: `BITRIX24_WEBHOOK_URL` (full URL including trailing slash)

    ```typescript
    // Bitrix24 REST API client for NGO outreach tracking

    const WEBHOOK = process.env.BITRIX24_WEBHOOK_URL

    interface BitrixLead {
      TITLE: string
      NAME?: string
      PHONE?: Array<{ VALUE: string; VALUE_TYPE: string }>
      WEB?: Array<{ VALUE: string; VALUE_TYPE: string }>
      ADDRESS?: string
      COMMENTS?: string
      SOURCE_ID?: string
      SOURCE_DESCRIPTION?: string
    }

    interface BitrixActivity {
      OWNER_TYPE_ID: number    // 1 = lead
      OWNER_ID: string | number
      TYPE_ID: number          // 2 = call
      SUBJECT: string
      DESCRIPTION?: string
      DIRECTION: number        // 2 = outgoing
      COMMUNICATIONS: Array<{
        VALUE: string
        ENTITY_ID?: string | number
        ENTITY_TYPE_ID?: number
      }>
      COMPLETED: string        // 'Y' | 'N'
    }

    async function callBitrix<T>(method: string, params: Record<string, unknown>): Promise<T | null> {
      if (!WEBHOOK) {
        console.warn('[bitrix24] BITRIX24_WEBHOOK_URL not set — skipping')
        return null
      }
      try {
        const res = await fetch(`${WEBHOOK}${method}/`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(params),
          signal: AbortSignal.timeout(5000),
        })
        if (!res.ok) return null
        const json = await res.json() as { result: T }
        return json.result ?? null
      } catch (err) {
        console.error('[bitrix24] API error:', err)
        return null
      }
    }

    /**
     * Create a CRM Lead in Bitrix24 for an external NGO/shelter found nearby an incident.
     * Returns the lead ID or null if creation failed / webhook not configured.
     */
    export async function createNGOLead(params: {
      shelterName: string
      phone: string | null
      website: string | null
      address: string
      incidentId: string
      incidentDescription: string
      urgency: string
      lat: number
      lng: number
    }): Promise<string | null> {
      const fields: BitrixLead = {
        TITLE: `[RESCURE] Animal Emergency near ${params.shelterName}`,
        PHONE: params.phone ? [{ VALUE: params.phone, VALUE_TYPE: 'WORK' }] : undefined,
        WEB: params.website ? [{ VALUE: params.website, VALUE_TYPE: 'WORK' }] : undefined,
        ADDRESS: params.address,
        COMMENTS: `Animal emergency reported on RESCURE platform.\n\nIncident ID: ${params.incidentId}\nUrgency: ${params.urgency}\nLocation: ${params.lat}, ${params.lng}\n\nDescription: ${params.incidentDescription}`,
        SOURCE_ID: 'WEB',
        SOURCE_DESCRIPTION: 'RESCURE Auto-Outreach System',
      }

      const result = await callBitrix<number>('crm.lead.add', { fields })
      return result != null ? String(result) : null
    }

    /**
     * Log an outbound call activity on a Bitrix24 lead.
     * Used to record the automated contact attempt.
     */
    export async function logCallActivity(params: {
      leadId: string
      shelterName: string
      phone: string
      incidentId: string
    }): Promise<boolean> {
      const activity: BitrixActivity = {
        OWNER_TYPE_ID: 1,   // Lead
        OWNER_ID: parseInt(params.leadId, 10),
        TYPE_ID: 2,         // Call
        SUBJECT: `Auto-contact: ${params.shelterName} — Incident ${params.incidentId}`,
        DESCRIPTION: `Automated outreach via RESCURE. An animal emergency was reported nearby. Please respond to incident ${params.incidentId}.`,
        DIRECTION: 2,       // Outgoing
        COMMUNICATIONS: [{ VALUE: params.phone }],
        COMPLETED: 'N',     // mark as pending — NGO hasn't confirmed yet
      }

      const result = await callBitrix<number>('crm.activity.add', { fields: activity })
      return result != null
    }
    ```
