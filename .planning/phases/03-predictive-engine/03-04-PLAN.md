    ```typescript
    'use client'

    import { useState } from 'react'
    import { toast } from 'sonner'

    export default function RunAnalysisBtn({ userRole }: { userRole: string }) {
      const [loading, setLoading] = useState(false)

      if (userRole !== 'PLATFORM_ADMIN') return null

      async function handleClick() {
        setLoading(true)
        try {
          const res = await fetch('/api/admin/risk/calculate', { method: 'POST' })
          if (res.ok) {
            const data = await res.json() as { zonesAnalyzed: number; surgesTriggered: number }
            toast.success(
              `Analysis complete — ${data.zonesAnalyzed} zone${data.zonesAnalyzed !== 1 ? 's' : ''} scanned, ` +
              `${data.surgesTriggered} surge${data.surgesTriggered !== 1 ? 's' : ''} triggered`
            )
          } else {
            toast.error('Analysis failed. Check server logs.')
          }
        } catch {
          toast.error('Network error — could not run analysis.')
        } finally {
          setLoading(false)
        }
      }

      return (
        <button
          onClick={handleClick}
          disabled={loading}
          className="rounded-lg px-3 py-1.5 text-sm font-medium bg-purple-600 text-white hover:bg-purple-700 disabled:opacity-50 transition-colors"
        >
          {loading ? 'Analyzing…' : '⚡ Run Predictive Analysis'}
        </button>
      )
    }
    ```

       ```typescript
       async function getRiskZones(): Promise<RiskZone[]> {
         const now = new Date()
         const zones = await prisma.coverageZone.findMany({
           select: { geohash: true },
           distinct: ['geohash'],
         })
         return Promise.all(
           zones.map(async (zone) => {
             const factors = await prisma.riskFactor.findMany({
               where: { geohash: zone.geohash, expiresAt: { gt: now } },
               select: { category: true, severity: true },
             })
             const riskScore = factors.reduce((sum, f) => sum + f.severity, 0.1)
             const [minLat, minLng, maxLat, maxLng] = decodeGeohash(zone.geohash)
             const hasActiveSurge =
               (await prisma.surgeEvent.count({
                 where: { geohash: zone.geohash, isActive: true },
               })) > 0
             return {
               geohash: zone.geohash,
               lat: (minLat + maxLat) / 2,
               lng: (minLng + maxLng) / 2,
               riskScore,
               hasActiveSurge,
               factors: factors.map((f) => ({ category: f.category, severity: f.severity })),
             }
           })
         )
       }
       ```

    ```typescript
    'use client'

    import { useEffect, useState } from 'react'
    import { MapContainer, TileLayer, CircleMarker, Popup, useMap } from 'react-leaflet'
    import L from 'leaflet'
    import 'leaflet/dist/leaflet.css'

    interface Hotspot {
      geohash: string; lat: number; lng: number; count: number; avgUrgency: number
    }
    interface RiskZone {
      geohash: string; lat: number; lng: number; riskScore: number
      hasActiveSurge: boolean; factors: { category: string; severity: number }[]
    }
    interface HotspotMapProps {
      hotspots: Hotspot[]
      days: number
      riskZones: RiskZone[]
      userRole: string
    }

    function urgencyColor(avg: number): string {
      if (avg >= 3.5) return '#ef4444'
      if (avg >= 2.5) return '#f97316'
      if (avg >= 1.5) return '#eab308'
      return '#22c55e'
    }

    function urgencyRadius(count: number, maxCount: number): number {
      const minR = 8, maxR = 30
      if (maxCount === 0) return minR
      return minR + ((count / maxCount) * (maxR - minR))
    }

    // Leaflet control rendered inside MapContainer via useMap()
    function MapLegend() {
      const map = useMap()
      useEffect(() => {
        const legend = L.control({ position: 'bottomleft' })
        legend.onAdd = () => {
          const div = L.DomUtil.create('div', 'leaflet-control leaflet-bar')
          const dot = (color: string) =>
            `<span style="display:inline-block;width:11px;height:11px;border-radius:50%;background:${color};margin-right:5px;vertical-align:middle"></span>`
          div.style.cssText =
            'background:white;padding:8px 12px;border-radius:6px;font-size:11px;line-height:1.9;box-shadow:0 1px 5px rgba(0,0,0,.2)'
          div.innerHTML = `
            <strong style="display:block;margin-bottom:4px;font-size:12px">Map Legend</strong>
            ${dot('#ef4444')}Historical — Critical<br>
            ${dot('#f97316')}Historical — High<br>
            ${dot('#eab308')}Historical — Medium<br>
            ${dot('#22c55e')}Historical — Low<br>
            ${dot('#9333ea')}Predictive — High Risk (&gt;0.6)<br>
            ${dot('#c084fc')}Predictive — Moderate (0.3–0.6)
          `
          return div
        }
        legend.addTo(map)
        return () => { legend.remove() }
      }, [map])
      return null
    }

    export default function HotspotMapInner({ hotspots, days, riskZones }: HotspotMapProps) {
      const [showHistorical, setShowHistorical] = useState(true)
      const [showPredictive, setShowPredictive] = useState(true)

      useEffect(() => {
        delete (L.Icon.Default.prototype as unknown as Record<string, unknown>)._getIconUrl
        L.Icon.Default.mergeOptions({
          iconRetinaUrl: '/leaflet/marker-icon-2x.png',
          iconUrl: '/leaflet/marker-icon.png',
          shadowUrl: '/leaflet/marker-shadow.png',
        })
      }, [])

      const center: [number, number] =
        hotspots.length > 0
          ? [hotspots[0].lat, hotspots[0].lng]
          : riskZones.length > 0
          ? [riskZones[0].lat, riskZones[0].lng]
          : [20.5937, 78.9629]

      const maxCount = Math.max(...hotspots.map((h) => h.count), 1)
      // Only show predictive circles with meaningful signal (>= 0.3)
      const visibleRisk = riskZones.filter((z) => z.riskScore >= 0.3)

      return (
        <div style={{ height: '100%', width: '100%', display: 'flex', flexDirection: 'column' }}>
          {/* Layer toggle — rendered above the map container */}
          <div
            style={{
              display: 'flex', alignItems: 'center', gap: '16px',
              padding: '8px 12px', background: 'white', borderBottom: '1px solid #e5e7eb',
              fontSize: '13px',
            }}
          >
            <span style={{ fontWeight: 500, color: '#6b7280' }}>Layers:</span>
            <label style={{ display: 'flex', alignItems: 'center', gap: '6px', cursor: 'pointer' }}>
              <input
                type="checkbox"
                checked={showHistorical}
                onChange={(e) => setShowHistorical(e.target.checked)}
                style={{ accentColor: '#ef4444' }}
              />
              <span>Historical</span>
            </label>
            <label style={{ display: 'flex', alignItems: 'center', gap: '6px', cursor: 'pointer' }}>
              <input
                type="checkbox"
                checked={showPredictive}
                onChange={(e) => setShowPredictive(e.target.checked)}
                style={{ accentColor: '#9333ea' }}
              />
              <span>Predictive</span>
            </label>
          </div>

          {/* Map fills remaining height */}
          <div style={{ flex: 1 }}>
            <MapContainer center={center} zoom={5} style={{ height: '100%', width: '100%' }}>
              <TileLayer
                attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>'
                url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
              />
              <MapLegend />

              {/* Historical layer — unchanged logic */}
              {showHistorical &&
                hotspots.map((hotspot) => (
                  <CircleMarker
                    key={`h-${hotspot.geohash}`}
                    center={[hotspot.lat, hotspot.lng]}
                    radius={urgencyRadius(hotspot.count, maxCount)}
                    pathOptions={{
                      color: urgencyColor(hotspot.avgUrgency),
                      fillColor: urgencyColor(hotspot.avgUrgency),
                      fillOpacity: 0.5 + (hotspot.count / maxCount) * 0.4,
                      weight: 1,
                    }}
                  >
                    <Popup>
                      <div className="min-w-[140px]">
                        <p className="font-semibold text-sm">{hotspot.count} incidents</p>
                        <p className="text-xs text-gray-500">in last {days} days</p>
                        <p className="text-xs text-gray-500">
                          Avg urgency: {hotspot.avgUrgency.toFixed(1)}/4
                        </p>
                      </div>
                    </Popup>
                  </CircleMarker>
                ))}

              {/* Predictive layer — purple circles scaled by riskScore */}
              {showPredictive &&
                visibleRisk.map((zone) => {
                  const color = zone.riskScore > 0.6 ? '#9333ea' : '#c084fc'
                  const radius = 8 + zone.riskScore * 18
                  const fillOpacity = 0.4 + zone.riskScore * 0.45
                  return (
                    <CircleMarker
                      key={`p-${zone.geohash}`}
                      center={[zone.lat, zone.lng]}
                      radius={radius}
                      pathOptions={{ color, fillColor: color, fillOpacity, weight: 1.5 }}
                    >
                      <Popup>
                        <div className="min-w-[160px]">
                          <p className="font-semibold text-sm" style={{ color: '#9333ea' }}>
                            Risk Score: {zone.riskScore.toFixed(2)}
                          </p>
                          {zone.hasActiveSurge && (
                            <p className="text-xs font-medium mt-0.5" style={{ color: '#ef4444' }}>
                              ⚡ Active Surge
                            </p>
                          )}
                          {zone.factors.length > 0 ? (
                            <ul className="mt-1 space-y-0.5">
                              {zone.factors.map((f, i) => (
                                <li key={i} className="text-xs text-gray-600">
                                  {f.category}: {f.severity.toFixed(2)}
                                </li>
                              ))}
                            </ul>
                          ) : (
                            <p className="text-xs text-gray-400 mt-1">No active risk factors</p>
                          )}
                        </div>
                      </Popup>
                    </CircleMarker>
                  )
                })}
            </MapContainer>
          </div>
        </div>
      )
    }
    ```
